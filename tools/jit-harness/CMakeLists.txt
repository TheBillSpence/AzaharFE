cmake_minimum_required(VERSION 3.25)
project(AzaharHostJitHarness LANGUAGES C CXX)
include(CTest)
set(DYNARMIC_FRONTENDS "A32" CACHE STRING "" FORCE)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT ON CACHE BOOL "" FORCE)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    message(FATAL_ERROR "Tests require a real Apple Silicon ARM64 macOS host.")
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../externals/dynarmic"
                 "${CMAKE_CURRENT_BINARY_DIR}/dynarmic" EXCLUDE_FROM_ALL)
add_executable(azahar_jit_harness main.cpp)
target_compile_features(azahar_jit_harness PRIVATE cxx_std_20)
target_link_libraries(azahar_jit_harness PRIVATE dynarmic)
foreach(test_case parity budget svc invalidate)
    add_test(NAME "azahar_jit_${test_case}" COMMAND azahar_jit_harness ${test_case})
    set_tests_properties("azahar_jit_${test_case}" PROPERTIES TIMEOUT 20)
endforeach()
